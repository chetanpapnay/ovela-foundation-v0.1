from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timedelta, timezone
from enum import StrEnum
from typing import Final
from uuid import UUID, uuid4


# Tolerance for clock differences between the issuer and the verifier.
# Contexts dated further in the future than this are rejected.
_MAX_CLOCK_SKEW: Final[timedelta] = timedelta(seconds=5)


class AuthorizationState(StrEnum):
    """
    Authentication state established by a trusted security subsystem.

    These values are security facts only when issued by the trusted
    authorization-context issuer. They must never be accepted directly
    from an untrusted request.
    """

    UNVERIFIED = "UNVERIFIED"
    AUTHENTICATED = "AUTHENTICATED"
    STEP_UP_VERIFIED = "STEP_UP_VERIFIED"


class AuthorizationContextError(ValueError):
    """Base error for invalid trusted authorization contexts."""


class AuthorizationContextExpired(AuthorizationContextError):
    """Raised when an authorization context is expired."""


class AuthorizationContextInvalid(AuthorizationContextError):
    """Raised when an authorization context violates its invariants."""


@dataclass(frozen=True, slots=True)
class AuthorizationContext:
    """
    Authorization context carrying security state and its provenance.

    The public constructor validates the structural shape of a context,
    but a structurally valid object is not automatically trusted.

    Only the trusted issuer marks a context as issuer-authenticated.

    This distinction is deliberate:

        structurally valid != trusted authorization authority

    The authorization boundary must require both:
    - valid context invariants
    - trusted issuer provenance

    The eventual authentication/session subsystem must establish the
    underlying authentication evidence before the trusted issuer creates
    the context.
    """

    context_id: UUID
    session_id: UUID
    subject_id: UUID
    state: AuthorizationState
    issued_at: datetime
    expires_at: datetime

    # Private provenance marker.
    #
    # This is excluded from equality/repr so the external shape of the
    # context remains stable. Direct construction receives None and is
    # therefore structurally valid but untrusted. The trusted issuer
    # inserts the private singleton marker.
    _issuer_marker: object = field(
        default=None,
        repr=False,
        compare=False,
    )

    def __post_init__(self) -> None:
        if not isinstance(self.context_id, UUID):
            raise TypeError("context_id must be a UUID")

        if not isinstance(self.session_id, UUID):
            raise TypeError("session_id must be a UUID")

        if not isinstance(self.subject_id, UUID):
            raise TypeError("subject_id must be a UUID")

        if not isinstance(self.state, AuthorizationState):
            raise TypeError("state must be an AuthorizationState")

        if not isinstance(self.issued_at, datetime):
            raise TypeError("issued_at must be a datetime")

        if not isinstance(self.expires_at, datetime):
            raise TypeError("expires_at must be a datetime")

        if self.issued_at.tzinfo is None:
            raise AuthorizationContextInvalid(
                "issued_at must be timezone-aware"
            )

        if self.expires_at.tzinfo is None:
            raise AuthorizationContextInvalid(
                "expires_at must be timezone-aware"
            )

        issued_at_utc = self.issued_at.astimezone(timezone.utc)
        expires_at_utc = self.expires_at.astimezone(timezone.utc)

        if expires_at_utc <= issued_at_utc:
            raise AuthorizationContextInvalid(
                "expires_at must be later than issued_at"
            )

    def is_trusted(self) -> bool:
        """
        Return whether this context carries trusted issuer provenance.

        A structurally valid AuthorizationContext is not sufficient for
        authorization. Only the trusted issuer can attach the internal
        provenance marker.

        This method does not grant authority; it only exposes the
        provenance state to the authorization boundary.
        """
        return self._issuer_marker is _INTERNAL_CONTEXT_MARKER

    def is_expired(self, now: datetime) -> bool:
        """
        Determine whether this context is expired at the supplied time.

        The caller must provide an explicit time so authorization decisions
        remain deterministic and testable.
        """
        if not isinstance(now, datetime):
            raise TypeError("now must be a datetime")

        if now.tzinfo is None:
            raise AuthorizationContextInvalid(
                "now must be timezone-aware"
            )

        return now.astimezone(timezone.utc) >= self.expires_at.astimezone(
            timezone.utc
        )

    def is_not_yet_valid(self, now: datetime) -> bool:
        """
        Return whether this context is dated in the future.

        A small clock-skew tolerance is allowed so a context issued a
        moment ago is not rejected because of minor clock differences.
        """
        if not isinstance(now, datetime):
            raise TypeError("now must be a datetime")

        if now.tzinfo is None:
            raise AuthorizationContextInvalid(
                "now must be timezone-aware"
            )

        return (
            self.issued_at.astimezone(timezone.utc)
            > now.astimezone(timezone.utc) + _MAX_CLOCK_SKEW
        )

    def require_valid_at(self, now: datetime) -> None:
        """
        Fail closed if the context is expired or dated in the future.
        """
        if self.is_expired(now):
            raise AuthorizationContextExpired(
                "authorization context has expired"
            )

        if self.is_not_yet_valid(now):
            raise AuthorizationContextInvalid(
                "authorization context is not yet valid"
            )

    def require_trusted(self) -> None:
        """
        Fail closed if the context was not issued by the trusted issuer.
        """
        if not self.is_trusted():
            raise AuthorizationContextInvalid(
                "authorization context lacks trusted issuer provenance"
            )

    def permits_authenticated(self) -> bool:
        """
        Return whether the context represents authenticated access.
        """
        return self.state in {
            AuthorizationState.AUTHENTICATED,
            AuthorizationState.STEP_UP_VERIFIED,
        }

    def permits_step_up(self) -> bool:
        """
        Return whether the context represents completed step-up
        verification.
        """
        return self.state is AuthorizationState.STEP_UP_VERIFIED


@dataclass(frozen=True, slots=True)
class _IssuerCapability:
    """
    Internal capability required to issue an AuthorizationContext.

    The capability is intentionally not representable by a caller-provided
    string, boolean, UUID, or enum value.

    A future authentication/session subsystem will own the issuer and the
    evidence required to obtain this capability.
    """

    nonce: UUID


_INTERNAL_ISSUER_CAPABILITY: Final[_IssuerCapability] = _IssuerCapability(
    nonce=uuid4(),
)


# Separate opaque provenance marker.

# A context is trusted only when this exact in-process object identity is
# attached by the issuer. A caller cannot manufacture an equivalent marker
# through ordinary values such as strings, UUIDs, booleans, or enums.
#
# This is a process-local capability boundary. It is not intended to defend
# against arbitrary code execution inside the trusted Python process itself.
_INTERNAL_CONTEXT_MARKER: Final[object] = object()


class TrustedAuthorizationContextIssuer:
    """
    Narrow issuance boundary for trusted authorization contexts.

    This is deliberately separate from ActionRequest.

    The current implementation establishes the structural and provenance
    boundary. A production authentication subsystem must become the sole
    legitimate caller that possesses the issuer capability and can establish
    the underlying authentication evidence.

    No request payload, external content, threat assessment, or conversational
    statement can manufacture this capability.
    """

    __slots__ = ()

    @staticmethod
    def _validate_capability(
        capability: _IssuerCapability,
    ) -> None:
        if not isinstance(capability, _IssuerCapability):
            raise PermissionError(
                "trusted issuer capability is required"
            )

        if capability is not _INTERNAL_ISSUER_CAPABILITY:
            raise PermissionError(
                "invalid trusted issuer capability"
            )

    @classmethod
    def issue(
        cls,
        *,
        capability: _IssuerCapability,
        session_id: UUID,
        subject_id: UUID,
        state: AuthorizationState,
        issued_at: datetime,
        expires_at: datetime,
    ) -> AuthorizationContext:
        """
        Issue a trusted authorization context.

        This method is intentionally capability-gated. It must not be exposed
        as a generic API endpoint or called from untrusted request handling.
        """
        cls._validate_capability(capability)

        return AuthorizationContext(
            context_id=uuid4(),
            session_id=session_id,
            subject_id=subject_id,
            state=state,
            issued_at=issued_at,
            expires_at=expires_at,
            _issuer_marker=_INTERNAL_CONTEXT_MARKER,
        )


def _issue_internal_authorization_context(
    *,
    session_id: UUID,
    subject_id: UUID,
    state: AuthorizationState,
    issued_at: datetime,
    expires_at: datetime,
) -> AuthorizationContext:
    """
    Internal application hook for the future authentication subsystem.

    This function is intentionally private. It exists so the authentication
    layer can later be introduced without allowing ActionRequest or external
    content to become an authorization authority.
    """
    return TrustedAuthorizationContextIssuer.issue(
        capability=_INTERNAL_ISSUER_CAPABILITY,
        session_id=session_id,
        subject_id=subject_id,
        state=state,
        issued_at=issued_at,
        expires_at=expires_at,
    )

from __future__ import annotations

class TrustClass(str, Enum):
    SAFE_FOR_DATA_EXTRACTION = "SAFE_FOR_DATA_EXTRACTION"
    REVIEW_REQUIRED = "REVIEW_REQUIRED"
    BLOCKED = "BLOCKED"


@dataclass(frozen=True)
class ContentAssessment:
    trust_class: TrustClass
    reasons: tuple[str, ...]
    contains_url: bool
    contains_payment_instruction: bool
    contains_privileged_instruction: bool
    contains_prompt_injection: bool
    contains_secret_request: bool


_URL_RE = re.compile(
    r"(?i)\b(?:https?://|www\.)[^\s<>'\"]+"
)


# Detect dangerous schemes even when attackers insert whitespace
# or normalized invisible characters between individual letters.
_DANGEROUS_SCHEME_RE = re.compile(
    r"(?i)"
    r"(?:"
    r"\bj\s*a\s*v\s*a\s*s\s*c\s*r\s*i\s*p\s*t\s*:"
    r"|\bv\s*b\s*s\s*c\s*r\s*i\s*p\s*t\s*:"
    r"|\bd\s*a\s*t\s*a\s*:"
    r"|\bf\s*i\s*l\s*e\s*:"
    r")"
)


_PAYMENT_RE = re.compile(
    r"(?i)"
    r"(?:"
    r"\btransfer\b"
    r"|\bsend\s+(?:money|payment|funds)\b"
    r"|\bpay\s+(?:₹|rs\.?|inr|\$|usd)?\s*\d"
    r"|\bpayment\b"

)

def _has_match(
    pattern: re.Pattern[str],
    text: str,
) -> bool:
    return pattern.search(text) is not None


def _contains_percent_encoding(text: str) -> bool:
    return _PERCENT_ESCAPE_RE.search(text) is not None


def _contains_invalid_percent_encoding(text: str) -> bool:
    return _INVALID_PERCENT_ESCAPE_RE.search(text) is not None


def _remove_invisible_characters(text: str) -> str:
    """
    Replace Unicode control and formatting characters with spaces
    in the detection representation.

    This preserves word boundaries across:
        - tab
        - carriage return
        - line feed
        - zero-width space
        - zero-width joiner/non-joiner
        - bidirectional formatting controls

    The original content is never modified.
    """

    return "".join(
        " "
        if unicodedata.category(character) in {"Cc", "Cf"}
        else character
        for character in text
    )


def _canonicalize_homoglyphs(text: str) -> str:
    """
    Convert common Unicode lookalikes into ASCII equivalents.

    Detection only. The original content remains untouched.
    """

    return text.translate(_HOMOGLYPH_TRANSLATION)


def _normalize_for_detection(
    content: str,
) -> tuple[str, bool]:
    """
    Normalize hostile content for detection only.

    The original input is never executed, opened, followed,
    interpreted as authority, or otherwise acted upon.

    Normalization is strictly bounded and includes:

    - Unicode NFKC normalization
    - HTML entity decoding
    - bounded percent decoding
    - replacement of invisible control/format characters
    - common Unicode homoglyph canonicalization
    """

    if len(content) > _MAX_CONTENT_LENGTH:
        return content, True

    current = content
    malformed_encoding = False

    for _ in range(_MAX_NORMALIZATION_PASSES):
        previous = current

        current = unicodedata.normalize(
            "NFKC",
            current,
        )

        current = html.unescape(current)

        if _contains_invalid_percent_encoding(current):
            malformed_encoding = True

        current = unquote(current)

        current = unicodedata.normalize(
            "NFKC",
            current,
        )

        current = _remove_invisible_characters(
            current
        )

        current = _canonicalize_homoglyphs(
            current
        )

        if current == previous:
            break

    return current, malformed_encoding


def assess_untrusted_content(
    content: str,
) -> ContentAssessment:
    """
    Classify external/untrusted content.

    Security invariants:

    - External content never grants authorization.
    - External content never grants privilege.
    - External content never authenticates a user.
    - Instructions inside external content are never executed.
    - URLs are detected but never followed.
    - Dangerous URL schemes are blocked.
    - Encoded dangerous content is normalized before classification.
    - Invisible-character obfuscation is normalized before classification.
    - Unicode homoglyph obfuscation is normalized before classification.
    - Malformed encoding fails closed into review.
    - Normalization has a strict upper bound.
    """

    if not isinstance(content, str):
        raise TypeError("content must be a string")

    original = content.strip()

    if not original:
        return ContentAssessment(
            trust_class=TrustClass.SAFE_FOR_DATA_EXTRACTION,
            reasons=("EMPTY_CONTENT",),
            contains_url=False,
            contains_payment_instruction=False,
            contains_privileged_instruction=False,
            contains_prompt_injection=False,
            contains_secret_request=False,
        )

    normalized, malformed_encoding = _normalize_for_detection(
        original
    )

    contains_url = _has_match(
        _URL_RE,
        normalized,
    )

    contains_dangerous_scheme = _has_match(
        _DANGEROUS_SCHEME_RE,
        normalized,
    )

    contains_payment = _has_match(
        _PAYMENT_RE,
        normalized,
    )

    contains_privileged = _has_match(
        _PRIVILEGED_RE,
        normalized,
    )

    contains_injection = _has_match(
        _PROMPT_INJECTION_RE,
        normalized,
    )

    contains_secret = _has_match(
        _SECRET_RE,
        normalized,
    )

    contains_dangerous_action = _has_match(
        _DANGEROUS_ACTION_RE,
        normalized,
    )

    original_contains_encoding = (
        _contains_percent_encoding(original)
        or original != normalized
    )

    reasons: list[str] = []

    if contains_url:
        reasons.append("URL_PRESENT")

    if contains_dangerous_scheme:
        reasons.append("DANGEROUS_URL_SCHEME")

    if malformed_encoding:
        reasons.append(
            "MALFORMED_OR_SUSPICIOUS_ENCODING"
        )

    if original_contains_encoding:
        reasons.append(
            "ENCODED_OR_NORMALIZED_CONTENT"
        )

    if contains_payment:
        reasons.append(
            "PAYMENT_OR_FINANCIAL_INSTRUCTION"
        )

    if contains_privileged:
        reasons.append(
            "PRIVILEGED_SECURITY_INSTRUCTION"
        )

    if contains_injection:
        reasons.append(
            "PROMPT_INJECTION_PATTERN"
        )

    if contains_secret:
        reasons.append(
            "SECRET_OR_CREDENTIAL_REQUEST"
        )

    if contains_dangerous_action:
        reasons.append(
            "IRREVERSIBLE_ACTION_LANGUAGE"
        )

    # Highest-risk classification wins.
    #
    # Dangerous executable/local-resource schemes are BLOCKED.
    # Malformed encoding is REVIEW_REQUIRED.
    # Security/credential/payment/injection language is REVIEW_REQUIRED.
    #
    # Nothing here grants authority.

    if contains_dangerous_scheme:
        trust_class = TrustClass.BLOCKED

    elif malformed_encoding:
        trust_class = TrustClass.REVIEW_REQUIRED

    elif contains_injection:
        trust_class = TrustClass.REVIEW_REQUIRED

    elif contains_secret:
        trust_class = TrustClass.REVIEW_REQUIRED

    elif contains_privileged:
        trust_class = TrustClass.REVIEW_REQUIRED

    elif contains_payment:
        trust_class = TrustClass.REVIEW_REQUIRED

    elif contains_dangerous_action:
        trust_class = TrustClass.REVIEW_REQUIRED

    else:
        trust_class = TrustClass.SAFE_FOR_DATA_EXTRACTION

    if not reasons:
        reasons.append(
            "NO_KNOWN_AUTHORITY_ESCALATION_PATTERN"
        )

    return ContentAssessment(
        trust_class=trust_class,
        reasons=tuple(reasons),
        contains_url=contains_url,
        contains_payment_instruction=contains_payment,
        contains_privileged_instruction=contains_privileged,
        contains_prompt_injection=contains_injection,
        contains_secret_request=contains_secret,
    )


def is_authorization_granting_content(
    content: str,
) -> bool:
    """
    Hard security boundary.

    No externally supplied content can ever grant
    authorization.

    Always returns False by design.
    """

    return False


def is_safe_to_use_as_authority(
    content: str,
) -> bool:
    """
    Hard security boundary.

    External content may be evidence, but it can never
    become security authority.

    Always returns False by design.
    """

    return False

"""
OVELA / JURIX realtime WebSocket transport boundary.

Endpoint:
    /v1/jurix/stream

Security model
--------------
This module is a transport boundary only.

Browser-controlled data is NEVER treated as trusted authorization
information. In particular, the client cannot establish:

- AuthorizationState
- subject identity
- session identity
- step-up verification
- approval evidence
- OVELA privileges
- internal API credentials

A trusted realtime session must be resolved by a server-side
RealtimeSessionResolver implementation.

The resolver is responsible for converting an authenticated transport
session into a trusted AuthorizationContext.

The conversation engine is also injected rather than guessed here.
This keeps the realtime transport independent from the security kernel,
voice provider, model provider, and future JURIX orchestration engine.

If either dependency is not configured, the endpoint fails closed with
a protocol-level error instead of fabricating authority or pretending
that conversation execution succeeded.
"""

MAX_TEXT_MESSAGE_BYTES = 64 * 1024
MAX_JSON_MESSAGE_BYTES = 128 * 1024

# 20 MiB is intentionally a hard upper bound for one realtime connection's
# accumulated inbound audio buffer. The normal client should stream much
# smaller chunks and terminate each utterance with audio_end.
MAX_AUDIO_BUFFER_BYTES = 20 * 1024 * 1024

# Prevent a client from keeping a socket alive forever without activity.
IDLE_TIMEOUT_SECONDS = 120

# Maximum number of protocol messages accepted per rolling second.
MAX_MESSAGES_PER_SECOND = 100

# Maximum number of concurrent response turns on one connection.
# JURIX currently operates as a single conversational turn stream.
MAX_CONCURRENT_TURNS = 1

# Binary audio frames are expected to remain reasonably small. This protects
# the event loop from a malicious client sending enormous WebSocket frames.
MAX_AUDIO_FRAME_BYTES = 256 * 1024

# The initial realtime protocol version.
PROTOCOL_VERSION = 1

class ClientMessageType(StrEnum):
    HELLO = "hello"
    TEXT = "text"
    AUDIO_START = "audio_start"
    AUDIO_END = "audio_end"
    INTERRUPT = "interrupt"
    PING = "ping"
    CLOSE = "close"


class ServerMessageType(StrEnum):
    READY = "ready"
    TRANSCRIPT = "transcript"
    RESPONSE_START = "response_start"
    RESPONSE_DELTA = "response_delta"
    AUDIO = "audio"
    RESPONSE_DONE = "response_done"
    INTERRUPTED = "interrupted"
    ERROR = "error"
    PONG = "pong"
    CLOSED = "closed"


class RealtimeErrorCode(StrEnum):
    PROTOCOL_ERROR = "protocol_error"
    MESSAGE_TOO_LARGE = "message_too_large"
    INVALID_MESSAGE = "invalid_message"
    INVALID_SEQUENCE = "invalid_sequence"
    AUTHENTICATION_REQUIRED = "authentication_required"
    SESSION_INVALID = "session_invalid"
    RUNTIME_UNAVAILABLE = "runtime_unavailable"
    RATE_LIMITED = "rate_limited"
    AUDIO_INVALID = "audio_invalid"
    INTERNAL_ERROR = "internal_error"


class RealtimeSessionResolver(Protocol):
    """
    Server-side authority for resolving a realtime transport session.

    IMPORTANT:
        The implementation MUST obtain the identity/session from the
        server-side authenticated transport/session layer.

    It MUST NOT trust arbitrary session IDs or authorization state supplied
    inside the WebSocket JSON payload.
    """

    async def resolve(
        self,
        websocket: WebSocket,
    ) -> AuthorizationContext | None:
        ...


class RealtimeConversationEngine(Protocol):
    """
    JURIX conversational runtime.

    The implementation owns model/provider orchestration.

    The realtime transport does not grant authorization or tool authority.
    """

    async def stream(
        self,
        *,
        authorization_context: AuthorizationContext,
        text: str | None,
        audio: bytes | None,
        conversation_id: UUID,
        turn_id: UUID,
    ) -> AsyncIterator["RealtimeOutput"]:
        ...

    async def interrupt(
        self,
        *,
        authorization_context: AuthorizationContext,
        conversation_id: UUID,
        turn_id: UUID | None,
    ) -> None:
        ...

class RealtimeOutputType(StrEnum):
    TRANSCRIPT = "transcript"
    RESPONSE_START = "response_start"
    RESPONSE_DELTA = "response_delta"
    AUDIO = "audio"
    RESPONSE_DONE = "response_done"


@dataclass(frozen=True, slots=True)
class RealtimeOutput:
    """
    Output emitted by the conversation engine.

    Only the transport-safe fields are exposed to the browser.

    AUDIO must contain raw bytes and is encoded by this transport as base64.
    """

    type: RealtimeOutputType
    text: str | None = None
    audio: bytes | None = None
    metadata: Mapping[str, Any] | None = None


# ---------------------------------------------------------------------------
# Global runtime registration
# ---------------------------------------------------------------------------

_session_resolver: RealtimeSessionResolver | None = None
_conversation_engine: RealtimeConversationEngine | None = None


def configure_realtime_runtime(
    *,
    session_resolver: RealtimeSessionResolver,
    conversation_engine: RealtimeConversationEngine,
) -> None:
    """
    Register the verified server-side realtime dependencies.

    This function deliberately requires both dependencies.

    There is no default/mock/fallback resolver because a fallback would risk
    turning an unauthenticated WebSocket into a trusted session.
    """

    if session_resolver is None:
        raise TypeError("session_resolver is required")

    if conversation_engine is None:
        raise TypeError("conversation_engine is required")

    global _session_resolver
    global _conversation_engine

    _session_resolver = session_resolver
    _conversation_engine = conversation_engine


def clear_realtime_runtime() -> None:
    """
    Clear registered realtime dependencies.

    Primarily useful for controlled application lifecycle and tests.
    """

    global _session_resolver
    global _conversation_engine

    _session_resolver = None
    _conversation_engine = None


# ---------------------------------------------------------------------------
# Connection state
# ---------------------------------------------------------------------------


@dataclass(slots=True)
class _RateWindow:
    started_at: float
    count: int


@dataclass(slots=True)
class _RealtimeConnection:
    websocket: WebSocket
    connection_id: UUID
    conversation_id: UUID
    authorization_context: AuthorizationContext

    hello_received: bool = False
    closed: bool = False

    audio_buffer: bytearray | None = None
    audio_active: bool = False

    active_turn_id: UUID | None = None
    active_turn_task: asyncio.Task[None] | None = None

    last_activity: float = 0.0
    rate_window: _RateWindow | None = None

    expected_sequence: int = 0

    def touch(self) -> None:
        self.last_activity = time.monotonic()

    def check_rate(self) -> bool:
        now = time.monotonic()

        if self.rate_window is None:
            self.rate_window = _RateWindow(
                started_at=now,
                count=1,
            )
            return True

        elapsed = now - self.rate_window.started_at

        if elapsed >= 1.0:
            self.rate_window = _RateWindow(
                started_at=now,
                count=1,
            )
            return True

        if self.rate_window.count >= MAX_MESSAGES_PER_SECOND:
            return False

        self.rate_window.count += 1
        return True


# ---------------------------------------------------------------------------
# Protocol encoding
# ---------------------------------------------------------------------------


def _json_bytes(payload: Mapping[str, Any]) -> bytes:
    return json.dumps(
        payload,
        ensure_ascii=False,
        separators=(",", ":"),
    ).encode("utf-8")


def _safe_metadata(
    metadata: Mapping[str, Any] | None,
) -> dict[str, Any]:
    """
    Return metadata safe for browser transport.

    Metadata is intentionally treated as informational only.

    The server must never expose secrets, authorization objects, internal
    capability objects, provider credentials, or arbitrary Python objects.
    """

    if not metadata:
        return {}

    result: dict[str, Any] = {}

    for key, value in metadata.items():
        if not isinstance(key, str):
            continue

        if len(key) > 128:
            continue

        if value is None or isinstance(
            value,
            (str, int, float, bool),
        ):
            result[key] = value

    return result


async def _send_json(
    websocket: WebSocket,
    payload: Mapping[str, Any],
) -> None:
    if websocket.client_state != WebSocketState.CONNECTED:
        return

    encoded = _json_bytes(payload)

    if len(encoded) > MAX_JSON_MESSAGE_BYTES:
        raise RuntimeError("server generated oversized realtime message")

    await websocket.send_text(encoded.decode("utf-8"))


async def _send_error(
    websocket: WebSocket,
    *,
    code: RealtimeErrorCode,
    message: str,
    retryable: bool = False,
) -> None:
    """
    Send a deliberately bounded protocol error.

    Internal exception details are never sent to the browser.
    """

    try:
        await _send_json(
            websocket,
            {
                "type": ServerMessageType.ERROR.value,
                "protocol_version": PROTOCOL_VERSION,
                "code": code.value,
                "message": message,
                "retryable": retryable,
            },
        )
    except Exception:
        logger.debug(
            "failed to send realtime error",
            exc_info=True,
        )


# ---------------------------------------------------------------------------
# Input validation
# ---------------------------------------------------------------------------


def _decode_json_message(raw: str) -> dict[str, Any]:
    encoded = raw.encode("utf-8")

    if len(encoded) > MAX_JSON_MESSAGE_BYTES:
        raise ValueError("message exceeds protocol size limit")

    try:
        decoded = json.loads(raw)
    except json.JSONDecodeError as exc:
        raise ValueError("invalid JSON message") from exc

    if not isinstance(decoded, dict):
        raise ValueError("realtime message must be a JSON object")

    return decoded


def _message_type(message: Mapping[str, Any]) -> str:
    value = message.get("type")

    if not isinstance(value, str):
        raise ValueError("message type is required")

    value = value.strip().lower()

    if not value:
        raise ValueError("message type is required")

    return value


def _validate_sequence(
    connection: _RealtimeConnection,
    message: Mapping[str, Any],
) -> None:
    """
    Validate optional client sequence numbers.

    A client that supplies a sequence number must advance monotonically.

    The server does not require clients to provide one yet, preserving
    compatibility with the first browser implementation.
    """

    if "sequence" not in message:
        return

    sequence = message["sequence"]

    if (
        isinstance(sequence, bool)
        or not isinstance(sequence, int)
        or sequence < 0
    ):
        raise ValueError("sequence must be a non-negative integer")

    if sequence != connection.expected_sequence:
        raise ValueError("invalid realtime message sequence")

    connection.expected_sequence += 1


def _validate_text_input(message: Mapping[str, Any]) -> str:
    value = message.get("text")

    if not isinstance(value, str):
        raise ValueError("text must be a string")

    text = value.strip()

    if not text:
        raise ValueError("text cannot be empty")

    if len(text.encode("utf-8")) > MAX_TEXT_MESSAGE_BYTES:
        raise ValueError("text exceeds protocol size limit")

    return text


def _decode_audio_base64(message: Mapping[str, Any]) -> bytes:
    value = message.get("audio")

    if not isinstance(value, str):
        raise ValueError("audio must be a base64 string")

    try:
        audio = base64.b64decode(
            value,
            validate=True,
        )
    except (ValueError, binascii.Error) as exc:
        raise ValueError("audio is not valid base64") from exc

    if not audio:
        raise ValueError("audio payload is empty")

    if len(audio) > MAX_AUDIO_FRAME_BYTES:
        raise ValueError("audio frame exceeds protocol size limit")

    return audio


# ---------------------------------------------------------------------------
# Session/runtime helpers
# ---------------------------------------------------------------------------


async def _resolve_session(
    websocket: WebSocket,
) -> AuthorizationContext:
    resolver = _session_resolver

    if resolver is None:
        raise RuntimeError("realtime session resolver is not configured")

    context = await resolver.resolve(websocket)

    if context is None:
        raise PermissionError("authenticated realtime session is required")

    if not isinstance(context, AuthorizationContext):
        raise PermissionError("realtime session resolver returned invalid context")

    # This is an independent defense-in-depth check.
    context.require_trusted()

    return context


async def _call_interrupt(
    connection: _RealtimeConnection,
) -> None:
    engine = _conversation_engine

    if engine is None:
        return

    try:
        await engine.interrupt(
            authorization_context=connection.authorization_context,
            conversation_id=connection.conversation_id,
            turn_id=connection.active_turn_id,
        )
    except Exception:
        # Interrupt is best-effort transport control. The exception is not
        # surfaced to the browser because doing so would expose implementation
        # details. The active task is still cancelled locally.
        logger.warning(
            "JURIX realtime interrupt failed",
            extra={
                "connection_id": str(connection.connection_id),
                "conversation_id": str(connection.conversation_id),
            },
            exc_info=True,
        )


async def _cancel_active_turn(
    connection: _RealtimeConnection,
) -> None:
    task = connection.active_turn_task

    if task is not None and not task.done():
        task.cancel()

        try:
            await task
        except asyncio.CancelledError:
            pass

    connection.active_turn_task = None
    connection.active_turn_id = None


# ---------------------------------------------------------------------------
# Conversation streaming
# ---------------------------------------------------------------------------


async def _stream_turn(
    connection: _RealtimeConnection,
    *,
    text: str | None,
    audio: bytes | None,
) -> None:
    engine = _conversation_engine

    if engine is None:
        await _send_error(
            connection.websocket,
            code=RealtimeErrorCode.RUNTIME_UNAVAILABLE,
            message="JURIX realtime runtime is unavailable.",
            retryable=True,
        )
        return

    if connection.active_turn_task is not None:
        raise RuntimeError("a realtime turn is already active")

    turn_id = uuid4()

    connection.active_turn_id = turn_id

    try:
        await _send_json(
            connection.websocket,
            {
                "type": ServerMessageType.RESPONSE_START.value,
                "protocol_version": PROTOCOL_VERSION,
                "conversation_id": str(connection.conversation_id),
                "turn_id": str(turn_id),
            },
        )

        async for output in engine.stream(
            authorization_context=connection.authorization_context,
            text=text,
            audio=audio,
            conversation_id=connection.conversation_id,
            turn_id=turn_id,
        ):
            if connection.closed:
                return

            if not isinstance(output, RealtimeOutput):
                logger.error(
                    "realtime engine emitted invalid output type",
                    extra={
                        "connection_id": str(
                            connection.connection_id
                        )
                    },
                )
                await _send_error(
                    connection.websocket,
                    code=RealtimeErrorCode.INTERNAL_ERROR,
                    message="JURIX realtime runtime returned invalid output.",
                )
                return

            await _send_output(
                connection,
                output,
                turn_id=turn_id,
            )

    except asyncio.CancelledError:
        raise

    except Exception:
        logger.exception(
            "JURIX realtime turn failed",
            extra={
                "connection_id": str(connection.connection_id),
                "conversation_id": str(connection.conversation_id),
                "turn_id": str(turn_id),
            },
        )

        await _send_error(
            connection.websocket,
            code=RealtimeErrorCode.INTERNAL_ERROR,
            message="JURIX could not complete the realtime turn.",
            retryable=True,
        )

    finally:
        connection.active_turn_id = None


async def _send_output(
    connection: _RealtimeConnection,
    output: RealtimeOutput,
    *,
    turn_id: UUID,
) -> None:
    output_type = output.type

    if output_type == RealtimeOutputType.TRANSCRIPT:
        if not isinstance(output.text, str):
            raise ValueError("transcript output requires text")

        await _send_json(
            connection.websocket,
            {
                "type": ServerMessageType.TRANSCRIPT.value,
                "protocol_version": PROTOCOL_VERSION,
                "conversation_id": str(connection.conversation_id),
                "turn_id": str(turn_id),
                "text": output.text,
                "metadata": _safe_metadata(output.metadata),
            },
        )
        return

    if output_type == RealtimeOutputType.RESPONSE_START:
        await _send_json(
            connection.websocket,
            {
                "type": ServerMessageType.RESPONSE_START.value,
                "protocol_version": PROTOCOL_VERSION,
                "conversation_id": str(connection.conversation_id),
                "turn_id": str(turn_id),
                "metadata": _safe_metadata(output.metadata),
            },
        )
        return

    if output_type == RealtimeOutputType.RESPONSE_DELTA:
        if not isinstance(output.text, str):
            raise ValueError("response_delta output requires text")

        await _send_json(
            connection.websocket,
            {
                "type": ServerMessageType.RESPONSE_DELTA.value,
                "protocol_version": PROTOCOL_VERSION,
                "conversation_id": str(connection.conversation_id),
                "turn_id": str(turn_id),
                "text": output.text,
            },
        )
        return

    if output_type == RealtimeOutputType.AUDIO:
        if not isinstance(output.audio, bytes):
            raise ValueError("audio output requires bytes")

        if not output.audio:
            return

        if len(output.audio) > MAX_AUDIO_FRAME_BYTES:
            raise ValueError("engine emitted oversized audio frame")

        await _send_json(
            connection.websocket,
            {
                "type": ServerMessageType.AUDIO.value,
                "protocol_version": PROTOCOL_VERSION,
                "conversation_id": str(connection.conversation_id),
                "turn_id": str(turn_id),
                "audio": base64.b64encode(
                    output.audio
                ).decode("ascii"),
                "metadata": _safe_metadata(output.metadata),
            },
        )
        return

    if output_type == RealtimeOutputType.RESPONSE_DONE:
        await _send_json(
            connection.websocket,
            {
                "type": ServerMessageType.RESPONSE_DONE.value,
                "protocol_version": PROTOCOL_VERSION,
                "conversation_id": str(connection.conversation_id),
                "turn_id": str(turn_id),
                "metadata": _safe_metadata(output.metadata),
            },
        )
        return

    raise ValueError("unsupported realtime output type")


# ---------------------------------------------------------------------------
# Connection lifecycle
# ---------------------------------------------------------------------------


async def _send_ready(
    connection: _RealtimeConnection,
) -> None:
    await _send_json(
        connection.websocket,
        {
            "type": ServerMessageType.READY.value,
            "protocol_version": PROTOCOL_VERSION,
            "connection_id": str(connection.connection_id),
            "conversation_id": str(connection.conversation_id),
            "capabilities": {
                "text": True,
                "audio_input": True,
                "audio_output": True,
                "interrupt": True,
                "streaming": True,
            },
        },
    )


async def _handle_hello(
    connection: _RealtimeConnection,
    message: Mapping[str, Any],
) -> None:
    if connection.hello_received:
        raise ValueError("hello has already been received")

    protocol_version = message.get("protocol_version")

    if protocol_version is None:
        protocol_version = PROTOCOL_VERSION

    if (
        isinstance(protocol_version, bool)
        or not isinstance(protocol_version, int)
    ):
        raise ValueError("protocol_version must be an integer")

    if protocol_version != PROTOCOL_VERSION:
        raise ValueError("unsupported realtime protocol version")

    requested_conversation_id = message.get("conversation_id")

    if requested_conversation_id is not None:
        if not isinstance(requested_conversation_id, str):
            raise ValueError("conversation_id must be a UUID string")

        try:
            requested_uuid = UUID(requested_conversation_id)
        except ValueError as exc:
            raise ValueError("conversation_id must be a UUID string") from exc

        # The client may resume/reference a conversation only by UUID.
        # Conversation ownership remains a server-side concern. The current
        # transport deliberately does not infer authorization from this UUID.
        connection.conversation_id = requested_uuid

    connection.hello_received = True

    await _send_ready(connection)


async def _handle_audio_start(
    connection: _RealtimeConnection,
) -> None:
    if connection.audio_active:
        raise ValueError("audio input is already active")

    connection.audio_buffer = bytearray()
    connection.audio_active = True


async def _handle_audio_end(
    connection: _RealtimeConnection,
) -> None:
    if not connection.audio_active:
        raise ValueError("audio input is not active")

    audio = bytes(connection.audio_buffer or b"")

    connection.audio_buffer = None
    connection.audio_active = False

    if not audio:
        raise ValueError("audio input is empty")

    if len(audio) > MAX_AUDIO_BUFFER_BYTES:
        raise ValueError("audio input exceeds connection limit")

    await _start_turn(
        connection,
        text=None,
        audio=audio,
    )


async def _handle_binary_audio(
    connection: _RealtimeConnection,
    audio: bytes,
) -> None:
    if not connection.audio_active:
        raise ValueError("binary audio received outside audio_start")

    if len(audio) > MAX_AUDIO_FRAME_BYTES:
        raise ValueError("audio frame exceeds protocol size limit")

    if connection.audio_buffer is None:
        connection.audio_buffer = bytearray()

    if (
        len(connection.audio_buffer) + len(audio)
        > MAX_AUDIO_BUFFER_BYTES
    ):
        connection.audio_buffer = None
        connection.audio_active = False
        raise ValueError("audio input exceeds connection limit")

    connection.audio_buffer.extend(audio)


async def _start_turn(
    connection: _RealtimeConnection,
    *,
    text: str | None,
    audio: bytes | None,
) -> None:
    if connection.active_turn_task is not None:
        raise RuntimeError("another realtime turn is already active")

    task = asyncio.create_task(
        _stream_turn(
            connection,
            text=text,
            audio=audio,
        )
    )

    connection.active_turn_task = task

    # The task owns its lifecycle. Do not await it here because that would
    # block receipt of interrupt/input messages and defeat bidirectional
    # realtime operation.


async def _handle_text(
    connection: _RealtimeConnection,
    message: Mapping[str, Any],
) -> None:
    text = _validate_text_input(message)

    await _start_turn(
        connection,
        text=text,
        audio=None,
    )


async def _handle_interrupt(
    connection: _RealtimeConnection,
) -> None:
    await _call_interrupt(connection)
    await _cancel_active_turn(connection)

    await _send_json(
        connection.websocket,
        {
            "type": ServerMessageType.INTERRUPTED.value,
            "protocol_version": PROTOCOL_VERSION,
            "conversation_id": str(connection.conversation_id),
        },
    )


async def _handle_ping(
    connection: _RealtimeConnection,
) -> None:
    await _send_json(
        connection.websocket,
        {
            "type": ServerMessageType.PONG.value,
            "protocol_version": PROTOCOL_VERSION,
        },
    )

@router.websocket("/v1/jurix/stream")
async def jurix_stream(websocket: WebSocket) -> None:
    """
    Bidirectional JURIX realtime stream.

    Authentication occurs BEFORE the application accepts browser messages
    beyond the WebSocket handshake.

    The resolver is the only component allowed to establish the trusted
    AuthorizationContext.
    """

    connection: _RealtimeConnection | None = None

    try:
        # ------------------------------------------------------------------
        # 1. Resolve trusted session BEFORE accepting application traffic.
        # ------------------------------------------------------------------

        try:
            authorization_context = await _resolve_session(
                websocket
            )
        except PermissionError:
            await websocket.close(
                code=status.WS_1008_POLICY_VIOLATION,
                reason="authenticated realtime session required",
            )
            return
        except RuntimeError:
            await websocket.close(
                code=status.WS_1013_TRY_AGAIN_LATER,
                reason="realtime runtime unavailable",
            )
            return

        # ------------------------------------------------------------------
        # 2. Accept the WebSocket only after trusted session resolution.
        # ------------------------------------------------------------------

        await websocket.accept()

        connection = _RealtimeConnection(
            websocket=websocket,
            connection_id=uuid4(),
            conversation_id=uuid4(),
            authorization_context=authorization_context,
            last_activity=time.monotonic(),
        )

        # ------------------------------------------------------------------
        # 3. Wait for hello before normal traffic.
        # ------------------------------------------------------------------

        while not connection.closed:

            if (
                time.monotonic() - connection.last_activity
                > IDLE_TIMEOUT_SECONDS
            ):
                await _send_error(
                    websocket,
                    code=RealtimeErrorCode.INVALID_SEQUENCE,
                    message="realtime connection timed out due to inactivity",
                    retryable=True,
                )
                break

            try:
                raw_message = await asyncio.wait_for(
                    websocket.receive(),
                    timeout=IDLE_TIMEOUT_SECONDS,
                )
            except asyncio.TimeoutError:
                await _send_error(
                    websocket,
                    code=RealtimeErrorCode.INVALID_SEQUENCE,
                    message="realtime connection timed out due to inactivity",
                    retryable=True,
                )
                break

            connection.touch()

            if not connection.check_rate():
                await _send_error(
                    websocket,
                    code=RealtimeErrorCode.RATE_LIMITED,
                    message="realtime message rate limit exceeded",
                    retryable=True,
                )

                await websocket.close(
                    code=status.WS_1013_TRY_AGAIN_LATER,
                    reason="rate limit exceeded",
                )
                return

            # --------------------------------------------------------------
            # Binary audio frame.
            # --------------------------------------------------------------

            if raw_message.get("bytes") is not None:
                if not connection.hello_received:
                    await _send_error(
                        websocket,
                        code=RealtimeErrorCode.INVALID_SEQUENCE,
                        message="hello is required before audio input",
                    )
                    continue

                audio = raw_message["bytes"]

                if not isinstance(audio, bytes):
                    await _send_error(
                        websocket,
                        code=RealtimeErrorCode.AUDIO_INVALID,
                        message="invalid binary audio frame",
                    )
                    continue

                try:
                    await _handle_binary_audio(
                        connection,
                        audio,
                    )
                except ValueError as exc:
                    await _send_error(
                        websocket,
                        code=RealtimeErrorCode.AUDIO_INVALID,
                        message=str(exc),
                    )

                continue

            # --------------------------------------------------------------
            # Text protocol message.
            # --------------------------------------------------------------

            raw_text = raw_message.get("text")

            if raw_text is None:
                await _send_error(
                    websocket,
                    code=RealtimeErrorCode.INVALID_MESSAGE,
                    message="unsupported WebSocket message",
                )
                continue

            if not isinstance(raw_text, str):
                await _send_error(
                    websocket,
                    code=RealtimeErrorCode.INVALID_MESSAGE,
                    message="invalid WebSocket text message",
                )
                continue

            if len(raw_text.encode("utf-8")) > MAX_JSON_MESSAGE_BYTES:
                await _send_error(
                    websocket,
                    code=RealtimeErrorCode.MESSAGE_TOO_LARGE,
                    message="realtime message exceeds protocol size limit",
                )
                continue

            try:
                message = _decode_json_message(raw_text)
                _validate_sequence(
                    connection,
                    message,
                )
                message_type = _message_type(message)

                # ----------------------------------------------------------
                # Hello
                # ----------------------------------------------------------

                if message_type == ClientMessageType.HELLO.value:
                    await _handle_hello(
                        connection,
                        message,
                    )
                    continue

                # ----------------------------------------------------------
                # Every message after hello.
                # ----------------------------------------------------------

                if not connection.hello_received:
                    raise ValueError(
                        "hello is required before realtime messages"
                    )

                if message_type == ClientMessageType.TEXT.value:
                    await _handle_text(
                        connection,
                        message,
                    )
                    continue

                if message_type == ClientMessageType.AUDIO_START.value:
                    await _handle_audio_start(
                        connection,
                    )
                    continue

                if message_type == ClientMessageType.AUDIO_END.value:
                    await _handle_audio_end(
                        connection,
                    )
                    continue

                if message_type == ClientMessageType.INTERRUPT.value:
                    await _handle_interrupt(
                        connection,
                    )
                    continue

                if message_type == ClientMessageType.PING.value:
                    await _handle_ping(
                        connection,
                    )
                    continue

                if message_type == ClientMessageType.CLOSE.value:
                    connection.closed = True
                    break

                raise ValueError(
                    f"unsupported realtime message type: {message_type}"
                )

            except ValueError as exc:
                await _send_error(
                    websocket,
                    code=RealtimeErrorCode.INVALID_MESSAGE,
                    message=str(exc),
                )

            except RuntimeError as exc:
                await _send_error(
                    websocket,
                    code=RealtimeErrorCode.RUNTIME_UNAVAILABLE,
                    message=str(exc),
                    retryable=True,
                )

            except Exception:
                logger.exception(
                    "unexpected JURIX realtime protocol error",
                    extra={
                        "connection_id": str(
                            connection.connection_id
                        ),
                    },
                )

                await _send_error(
                    websocket,
                    code=RealtimeErrorCode.INTERNAL_ERROR,
                    message="JURIX realtime transport failed.",
                    retryable=True,
                )

    except WebSocketDisconnect:
        logger.info(
            "JURIX realtime client disconnected",
            extra={
                "connection_id": (
                    str(connection.connection_id)
                    if connection is not None
                    else None
                ),
            },
        )

    except Exception:
        logger.exception(
            "JURIX realtime connection failed",
            extra={
                "connection_id": (
                    str(connection.connection_id)
                    if connection is not None
                    else None
                ),
            },
        )

        if (
            websocket.client_state
            == WebSocketState.CONNECTED
        ):
            try:
                await _send_error(
                    websocket,
                    code=RealtimeErrorCode.INTERNAL_ERROR,
                    message="JURIX realtime transport failed.",
                    retryable=True,
                )
            finally:
                try:
                    await websocket.close(
                        code=status.WS_1011_INTERNAL_ERROR,
                        reason="internal realtime failure",
                    )
                except Exception:
                    pass

    finally:
        if connection is not None:
            connection.closed = True

            if connection.active_turn_task is not None:
                await _call_interrupt(connection)
                await _cancel_active_turn(connection)

            connection.audio_buffer = None
            connection.audio_active = False

            if (
                websocket.client_state
                == WebSocketState.CONNECTED
            ):
                try:
                    await _send_json(
                        websocket,
                        {
                            "type": ServerMessageType.CLOSED.value,
                            "protocol_version": PROTOCOL_VERSION,
                        },
                    )
                except Exception:
                    pass

"""
Ovela Action Execution Boundary.

This module is the boundary between an authorized ActionRequest
and a financial ledger side effect.

Security principles:
- Authorization must succeed before execution.
- Trusted AuthorizationContext is mandatory.
- ActionRequest.authorization_state is never treated as authority.
- Action session identity must match trusted authorization context.
- Only supported high-impact actions may execute.
- Payment intent must be complete and internally consistent.
- An action ID may execute at most once per executor instance.
- A failed authorization produces no ledger event.
- A successful payment produces exactly one ledger event.
- Caller-controlled input is captured once at the security boundary.
- The canonical request is converted to the exact ActionRequest type.
- Authorization and execution use separate independent snapshots.
- Authorization-side mutation cannot alter execution intent.
- External content has no authority over execution.
- Security-relevant time comes from the trusted temporal boundary.
- The execution timestamp is captured once per successful execution.
- Callers cannot supply ledger timestamps through execute().
"""

from __future__ import annotations

from dataclasses import dataclass
from threading import RLock

from .approval_evidence import (
    ApprovalEvidence,
    ApprovalEvidenceConsumed,
    ApprovalEvidenceExpired,
    ApprovalEvidenceInvalid,
    ApprovalEvidenceRegistry,
)
from .authorization_context import AuthorizationContext
from .ledger import AppendOnlyLedger, LedgerConflict
from .models import (
    ActionRequest,
    EvidenceStatus,
    EventType,
    FinancialEvent,
)

_DEFAULT_APPROVAL_REGISTRY = ApprovalEvidenceRegistry()


@dataclass(frozen=True)
class ExecutionResult:
    """
    Immutable result of an execution attempt.
    """

    executed: bool
    reason: str
    action_id: str


class ActionExecutor:
    """
    Execute security-sensitive actions against the ledger.

    The executor owns the single-use execution boundary.

    An action that has successfully produced a ledger event is marked
    as executed and cannot be executed again by this executor.

    Authorization authority is deliberately supplied through a trusted
    AuthorizationContext rather than through caller-controlled fields
    inside ActionRequest.

    Temporal authority is deliberately not accepted through execute().
    Financial event timestamps are resolved internally through the
    trusted temporal boundary.
    """

    def __init__(
        self,
        ledger: AppendOnlyLedger,
        approval_registry: ApprovalEvidenceRegistry | None = None,
    ):
        if not isinstance(ledger, AppendOnlyLedger):
            raise TypeError("ledger must be an AppendOnlyLedger")

        if approval_registry is None:
            approval_registry = _DEFAULT_APPROVAL_REGISTRY

        if type(approval_registry) is not ApprovalEvidenceRegistry:
            raise TypeError(
                "approval_registry must be an ApprovalEvidenceRegistry"
            )

        self._ledger = ledger
        self._approval_registry = approval_registry
        self._executed_actions: set[str] = set()
        self._lock = RLock()

    def execute(
        self,
        action: ActionRequest,
        authorization_context: AuthorizationContext,
        approval_evidence: ApprovalEvidence | None = None,
        *,
        risk_evaluation: RiskEvaluation | None = None,
        step_up_evidence: StepUpEvidence | None = None,
    ) -> ExecutionResult:
        """
        Authorize and execute one action.

        Security ordering:

            validate request
                ↓
            validate trusted authorization context
                ↓
            capture caller input once
                ↓
            normalize into exact ActionRequest boundary
                ↓
            create independent authorization/execution snapshots
                ↓
            single-use protection
                ↓
            authorize authorization snapshot against trusted context and approval
                ↓
            validate execution snapshot
                ↓
            resolve trusted temporal context
                ↓
            consume single-use trusted approval
                ↓
            append exactly one ledger event
                ↓
            mark action executed

        The caller-controlled ActionRequest is captured exactly once.

        The captured data is then reconstructed as the exact
        ActionRequest class. This prevents a caller-controlled subclass
        from overriding model_copy() or model_dump() and changing the
        transaction between the capture and validation boundary.

        Authorization and execution then receive separate deep copies.
        Therefore:

        - caller mutation cannot change the captured transaction
        - subclass model_copy() overrides cannot affect the boundary
        - subclass model_dump() overrides cannot affect the boundary
        - authorization mutation cannot affect execution
        - execution uses only the isolated execution snapshot

        Authorization authority comes exclusively from the supplied
        AuthorizationContext.

        The temporal boundary is also explicit:

        - production execution obtains time only through
          from_server_clock()
        - execute() accepts no caller-supplied timestamp
        - datetime.now() is never called directly by this module
        - one trusted timestamp is reused for event_date and recorded_at

        The lock covers the authorization-to-ledger boundary so two
        concurrent calls cannot execute the same action simultaneously.
        """

        if not isinstance(action, ActionRequest):
            return ExecutionResult(
                executed=False,
                reason="invalid action request",
                action_id="",
            )

        if not isinstance(
            authorization_context,
            AuthorizationContext,
        ):
            return ExecutionResult(
                executed=False,
                reason="trusted authorization context is missing or invalid",
                action_id="",
            )

        # --------------------------------------------------------------
        # 1. Capture the caller-controlled request exactly once.
        #
        # Call the base ActionRequest implementation directly so a
        # malicious or unexpected ActionRequest subclass cannot replace
        # this operation with its own model_copy() implementation.
        # --------------------------------------------------------------

        captured = ActionRequest.model_copy(
            action,
            deep=True,
        )

        # --------------------------------------------------------------
        # 2. Serialize using the trusted ActionRequest implementation.
        #
        # IMPORTANT:
        # Do NOT call captured.model_dump().
        #
        # captured may still be an ActionRequest subclass, and a malicious
        # subclass can override model_dump() to alter security-sensitive
        # values before canonical validation.
        #
        # Calling ActionRequest.model_dump(captured) invokes the trusted
        # base implementation directly and bypasses subclass overrides.
        #
        # warnings=False: unvalidated objects (model_construct()/
        # model_copy(update=...)) may hold invalid values. Pydantic would
        # emit a serializer UserWarning here. The real rejection happens
        # in model_validate() below, which raises ValidationError.
        # --------------------------------------------------------------

        captured_data = ActionRequest.model_dump(
            captured,
            warnings=False,
        )

        # --------------------------------------------------------------
        # 3. Reconstruct an exact ActionRequest.
        #
        # This is the intentional type-boundary normalization.
        #
        # Validation now occurs through the trusted ActionRequest model,
        # rejecting unvalidated model_construct()/model_copy() payloads.
        # --------------------------------------------------------------

        canonical_snapshot = ActionRequest.model_validate(
            captured_data,
        )

        # --------------------------------------------------------------
        # 4. Create two independent snapshots from the same canonical
        #    transaction intent.
        #
        # Both snapshots are now plain ActionRequest instances.
        # Authorization cannot mutate execution intent.
        # --------------------------------------------------------------

        authorization_snapshot = canonical_snapshot.model_copy(
            deep=True,
        )

        execution_snapshot = canonical_snapshot.model_copy(
            deep=True,
        )

        action_id = str(execution_snapshot.action_id)

        with self._lock:
            # ----------------------------------------------------------
            # 5. Single-use protection.
            # ----------------------------------------------------------

            if action_id in self._executed_actions:
                return ExecutionResult(
                    executed=False,
                    reason="action has already been executed",
                    action_id=action_id,
                )

            # ----------------------------------------------------------
            # 6. Authorization.
            #
            # Authorization receives:
            #    - isolated action intent
            #    - trusted authorization context
            #
            # ActionRequest.authorization_state is not authoritative.
            # ----------------------------------------------------------

            decision = authorize_action(
                authorization_snapshot,
                authorization_context,
                approval_evidence,
                risk_evaluation=risk_evaluation,
                step_up_evidence=step_up_evidence,
            )

            if not decision.allowed:
                return ExecutionResult(
                    executed=False,
                    reason=decision.reason,
                    action_id=action_id,
                )

            # ----------------------------------------------------------
            # 7. Execution currently supports payments only.
            #
            # IMPORTANT:
            # use execution_snapshot, never authorization_snapshot.
            # ----------------------------------------------------------

            if execution_snapshot.action_type != "PAYMENT_INITIATE":
                return ExecutionResult(
                    executed=False,
                    reason="action type is not executable",
                    action_id=action_id,
                )

            # ----------------------------------------------------------
            # 8. Complete payment intent.
            #
            # These checks are intentionally repeated at the execution
            # boundary. Authorization and execution are separate
            # security boundaries and must both fail closed.
            # ----------------------------------------------------------

            if (
                execution_snapshot.amount is None
                or execution_snapshot.amount <= 0
            ):
                return ExecutionResult(
                    executed=False,
                    reason="payment amount must be positive",
                    action_id=action_id,
                )

            if not execution_snapshot.amount.is_finite():
                return ExecutionResult(
                    executed=False,
                    reason="payment amount must be finite",
                    action_id=action_id,
                )

            if not execution_snapshot.currency:
                return ExecutionResult(
                    executed=False,
                    reason="payment currency is required",
                    action_id=action_id,
                )

            currency = execution_snapshot.currency.upper()

            if len(currency) != 3 or not currency.isalpha():
                return ExecutionResult(
                    executed=False,
                    reason="payment currency must be a three-letter code",
                    action_id=action_id,
                )

            if not execution_snapshot.destination_id:
                return ExecutionResult(
                    executed=False,
                    reason="payment destination is required",
                    action_id=action_id,
                )

            # Caller-controlled destination_verified, user_confirmation,
            # and confirmation_action_id are deliberately not consulted.
            # Approval authority comes only from trusted ApprovalEvidence.

            # ----------------------------------------------------------
            # 9. Resolve trusted temporal context.
            #
            # The caller cannot provide or influence the temporal
            # context through execute().
            #
            # The temporal boundary is responsible for obtaining the
            # authoritative server timestamp.
            #
            # One timestamp is captured and reused for the complete
            # financial event so event_date and recorded_at cannot drift.
            # ----------------------------------------------------------

            temporal_context = from_server_clock("UTC")

            event_timestamp = temporal_context.timestamp_utc

            try:
                self._approval_registry.consume(
                    approval_evidence,
                    now=event_timestamp,
                )
            except (
                ApprovalEvidenceConsumed,
                ApprovalEvidenceExpired,
                ApprovalEvidenceInvalid,
                TypeError,
                ValueError,
            ):
                return ExecutionResult(
                    executed=False,
                    reason="trusted approval evidence could not be consumed",
                    action_id=action_id,
                )

            # ----------------------------------------------------------
            # 10. Build the immutable financial event exclusively from
            #      the isolated execution snapshot and trusted temporal
            #      context.
            # ----------------------------------------------------------

            event = FinancialEvent(
                account_id=execution_snapshot.session_id,
                source_id="ACTION_EXECUTOR",
                source_event_id=action_id,
                event_type=EventType.EXPENSE,
                amount=execution_snapshot.amount,
                currency=currency,
                direction="DEBIT",
                event_date=event_timestamp,
                recorded_at=event_timestamp,
                evidence_status=EvidenceStatus.USER_CONFIRMED,
                note=f"Payment action {action_id}",
            )

            # ----------------------------------------------------------
            # 11. Commit exactly one ledger event.
            #
            # LedgerConflict is treated as a failed execution. The
            # action is NOT marked executed unless the ledger accepts it.
            # ----------------------------------------------------------

            try:
                self._ledger.append(event)
            except LedgerConflict:
                return ExecutionResult(
                    executed=False,
                    reason=(
                        "action could not be committed because "
                        "it was already recorded"
                    ),
                    action_id=action_id,
                )

            # ----------------------------------------------------------
            # 12. Only mark the action executed after the ledger commit.
            # ----------------------------------------------------------

            self._executed_actions.add(action_id)

            return ExecutionResult(
                executed=True,
                reason="action executed successfully",
                action_id=action_id,
            )

class ActionRequest(BaseModel):
    """
    Exact intent for a requested action.

    Security-sensitive fields are part of the action identity.

        frozen=True,

        min_length=1,
        max_length=128,

    currency: str | None = Field(
        default=None,
        min_length=3,
        max_length=3,
        pattern=r"^[A-Za-z]{3}$",
    )

    destination_id: str | None = Field(
        default=None,
        min_length=1,
        max_length=256,
    )

    user_confirmation: StrictBool = False

    confirmation_action_id: UUID | None = None

    destination_verified: StrictBool = False

    @field_validator("session_id", "action_type", "destination_id")
    @classmethod
    def validate_nonblank_text(
        cls,
        value: str | None,
    ) -> str | None:
        if value is None:
            return None

        if not value.strip():
            raise ValueError("value must not be blank")

        return value

    @field_validator("amount")
    @classmethod
    def validate_amount(
        cls,
        value: Decimal | None,
    ) -> Decimal | None:
        if value is None:
            return None

        if not value.is_finite():
            raise ValueError("amount must be finite")

        if value < 0:
            raise ValueError(
                "amount must be non-negative"
            )

        return value

    @field_validator("currency")
    @classmethod
    def normalize_currency(
        cls,
        value: str | None,
    ) -> str | None:
        if value is None:
            return None

        return value.upper()

    def model_copy(
        self,
        *,
        update: Mapping[str, Any] | None = None,
        deep: bool = False,
    ) -> "ActionRequest":
        """
        Copy the request while preserving immutable transaction identity.

        Pydantic's default model_copy(update=...) does not validate update
        values and can bypass frozen-field assignment protection.

        action_id is the transaction identity and therefore cannot be
        supplied through the update mechanism at all.

        if update is not None and "action_id" in update:
            raise TypeError(
                "action_id cannot be changed through model_copy()"
            )

        return super().model_copy(
            update=update,
            deep=deep,
        )

from __future__ import annotations

from threading import RLock
from uuid import UUID

from .models import FinancialEvent


class LedgerConflict(Exception):
    """Raised when an event violates ledger append-only invariants."""


class AppendOnlyLedger:
    """
    Thread-safe append-only ledger.

    Events may be appended exactly once according to their
    (source_id, source_event_id) identity.

    The ledger never exposes its internally stored model instances
    directly, preventing callers from mutating stored state after
    insertion.

    Security boundary:
    External FinancialEvent subclasses are treated as untrusted input.

    The ledger canonicalizes every incoming event into the exact
    FinancialEvent base type using trusted base-class serialization
    and validation. This prevents malicious subclasses from overriding
    model_copy() or model_dump() and changing ledger data during
    insertion or retrieval.
    """

    @staticmethod
    def _canonicalize_event(event: FinancialEvent) -> FinancialEvent:
        """
        Convert an incoming event into an exact FinancialEvent instance.

        Security rationale:

        Do NOT call:
            event.model_copy()
            event.model_dump()

        directly, because FinancialEvent may have been subclassed by
        caller-controlled code that overrides those methods.

        Calling the trusted FinancialEvent implementation directly
        bypasses subclass overrides.

        The serialized data is then validated again through the exact
        FinancialEvent model so the ledger stores only canonical,
        validated base-model instances.
        """

        event_data = FinancialEvent.model_dump(event)

        return FinancialEvent.model_validate(event_data)

    @classmethod
    def _copy_event(cls, event: FinancialEvent) -> FinancialEvent:
        """
        Return an independent exact FinancialEvent copy.

        The event is canonicalized first so no caller-controlled
        subclass implementation can alter the returned data.
        """

        canonical_event = cls._canonicalize_event(event)

        return FinancialEvent.model_copy(
            canonical_event,
            deep=True,
        )

    def __init__(self) -> None:
        self._events: list[FinancialEvent] = []
        self._source_keys: set[tuple[str, str]] = set()
        self._lock = RLock()

    def append(self, event: FinancialEvent) -> FinancialEvent:
        """
        Append an event exactly once.

        Duplicate or replayed source events are rejected atomically
        under the ledger lock.

        The incoming event is canonicalized before it becomes part
        of the ledger's internal state.
        """

        if not isinstance(event, FinancialEvent):
            raise TypeError("event must be a FinancialEvent")

        with self._lock:
            stored_event = self._canonicalize_event(event)

            key = (
                stored_event.source_id,
                stored_event.source_event_id,
            )

            if key in self._source_keys:
                raise LedgerConflict(
                    "duplicate or replayed source event"
                )

            self._source_keys.add(key)
            self._events.append(stored_event)

            return self._copy_event(stored_event)

    def all(self) -> list[FinancialEvent]:
        """
        Return an independent snapshot of all ledger events.

        Every returned object is an exact FinancialEvent base-model
        instance and cannot mutate internal ledger state.
        """

        with self._lock:
            return [
                self._copy_event(event)
                for event in self._events
            ]

    def get(self, event_id: UUID) -> FinancialEvent | None:
        """
        Return an independent copy of the requested event.

        Returns None when the event does not exist.
        """

        if not isinstance(event_id, UUID):
            raise TypeError("event_id must be a UUID")

        with self._lock:
            for event in self._events:
                if event.event_id == event_id:
                    return self._copy_event(event)

        return None
